cmake_minimum_required(VERSION 3.16)
project(DialysisLabBrowser LANGUAGES C CXX)
if(NOT EMSCRIPTEN)
  message(FATAL_ERROR "Configure using emcmake; this target requires WebAssembly")
endif()
get_filename_component(ROOT "${CMAKE_CURRENT_SOURCE_DIR}/../.." ABSOLUTE)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_RUNTIME_OUTPUT_DIRECTORY "${ROOT}/build/browser")
set(DEPS "${ROOT}/build/gui-deps")
set(SDL_SHARED OFF CACHE BOOL "" FORCE)
set(SDL_STATIC ON CACHE BOOL "" FORCE)
foreach(feature TEST TESTS AUDIO JOYSTICK HAPTIC HIDAPI SENSOR DBUS IBUS WAYLAND VULKAN KMSDRM LIBUDEV)
  set(SDL_${feature} OFF CACHE BOOL "" FORCE)
endforeach()
add_subdirectory("${DEPS}/SDL2-2.32.10" sdl EXCLUDE_FROM_ALL)
set(LV_BUILD_CONF_PATH "${ROOT}/src/ui/lv_conf.h" CACHE PATH "" FORCE)
foreach(feature CONFIG_LV_BUILD_EXAMPLES CONFIG_LV_BUILD_DEMOS CONFIG_LV_USE_THORVG CONFIG_LV_USE_THORVG_INTERNAL LV_USE_FIND_PACKAGE LV_USE_PKG_CONFIG LV_USE_FETCH_DEPENDENCIES LV_BUILD_INSTALL)
  set(${feature} OFF CACHE BOOL "" FORCE)
endforeach()
add_subdirectory("${DEPS}/lvgl-9.6.0" lvgl EXCLUDE_FROM_ALL)
add_executable(ui "${ROOT}/src/browser/ui.cpp" "${ROOT}/src/ui/font.cpp")
target_include_directories(ui PRIVATE "${ROOT}/src" "${DEPS}/lvgl-9.6.0/include")
target_link_libraries(ui PRIVATE lvgl SDL2-static)
target_compile_options(ui PRIVATE -O2 -fexceptions -fno-fast-math)
target_link_options(ui PRIVATE -O2 -fexceptions -sDISABLE_EXCEPTION_CATCHING=0 -sMODULARIZE=1 -sEXPORT_NAME=createUI -sALLOW_MEMORY_GROWTH=1 -sENVIRONMENT=web "-sEXPORTED_FUNCTIONS=['_main','_browser_view','_browser_result','_browser_disconnected','_browser_page','_browser_resize','_browser_click','_browser_primary','_browser_primary_label','_browser_dialog','_browser_rx_value','_browser_rx_enabled','_browser_review_rx']" "-sEXPORTED_RUNTIME_METHODS=['ccall','cwrap']")

add_executable(engine "${ROOT}/src/browser/engine.cpp")
target_compile_options(engine PRIVATE -O2 -fexceptions -fno-fast-math)
target_link_options(engine PRIVATE -O2 -fexceptions -sDISABLE_EXCEPTION_CATCHING=0 -sMODULARIZE=1 -sEXPORT_NAME=createEngine -sALLOW_MEMORY_GROWTH=1 "-sEXPORTED_FUNCTIONS=['_browser_reset','_browser_rpc']" "-sEXPORTED_RUNTIME_METHODS=['ccall','cwrap']")
